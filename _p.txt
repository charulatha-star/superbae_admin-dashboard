'use client';

import { useState, useEffect } from 'react';
import Link from 'next/link';
import { fetchApi } from '../../../lib/api/api';
import { AdminTableSkeleton } from '../../../components/admin/Skeleton';
import { NoData } from '../../../components/admin/NoData/NoData';
import { PermissionGate } from '../../../components/admin/PermissionGate';
import { ConfirmModal } from '../../../components/admin/ConfirmModal';
import { Toast } from '../../../components/admin/Toast';
import { Search, Plus, Edit2, Trash2, Send } from 'lucide-react';
import {
  canSend,
  channelLabel,
  describeSendOutcome,
  friendlyError,
  segmentLabel,
  statusBadgeModifier,
  type NotificationRecord,
  type NotificationSendResponse,
} from '../../../lib/notifications/helpers';
import { useNotificationOptions } from '../../../lib/notifications/options';
import styles from '../users/page.module.css';
import extra from './notifications.module.css';

type ToastState = { message: string; type: 'success' | 'error' | 'warning' | 'info' } | null;

/** Schedule cell: canonical scheduleAt, legacy scheduledFor fallback, else â€”. */
function formatSchedule(notification: NotificationRecord): string {
  const value = notification.scheduleAt ?? notification.scheduledFor;
  if (!value) return 'â€”';
  const date = new Date(value);
  return Number.isNaN(date.getTime()) ? 'â€”' : date.toLocaleString();
}

/** Delivery summary under the status badge â€” only safe sendResult fields. */
function deliveryLine(notification: NotificationRecord): string {
  const result = notification.sendResult;
  if (!result) return '';
  if (notification.status === 'sent') return `${result.successful ?? 0}/${result.targeted ?? 0} delivered`;
  if (notification.status === 'failed') return `0/${result.targeted ?? 0} delivered`;
  return '';
}

export default function NotificationsPage() {
  const [notifications, setNotifications] = useState<NotificationRecord[]>([]);
  const [loading, setLoading] = useState(true);
  const [loadError, setLoadError] = useState<string | null>(null);
  const [reloadKey, setReloadKey] = useState(0);
  const { options, loading: optionsLoading, error: optionsError, retry: retryOptions } = useNotificationOptions();
  const [search, setSearch] = useState('');
  const [currentPage, setCurrentPage] = useState(1);
  const [deleteId, setDeleteId] = useState<string | null>(null);
  const [sendTarget, setSendTarget] = useState<NotificationRecord | null>(null);
  const [sending, setSending] = useState(false);
  const [toast, setToast] = useState<ToastState>(null);

  // Initial load / explicit reloads. `loading` starts true and the first
  // setState only happens after the awaited fetch, so the effect body never
  // updates state synchronously.
  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const data = await fetchApi<NotificationRecord[]>('/notifications');
        if (!cancelled) {
          setNotifications(data);
          setLoadError(null);
        }
      } catch (error: unknown) {
        if (!cancelled) setLoadError(friendlyError(error, 'Failed to load notifications.'));
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [reloadKey]);

  const reload = (): void => {
    setLoading(true);
    setLoadError(null);
    setReloadKey((current) => current + 1);
  };

  const handleDelete = async (): Promise<void> => {
    if (!deleteId) return;
    try {
      await fetchApi(`/notifications/${deleteId}`, { method: 'DELETE' });
      setNotifications((prev) => prev.filter((entry) => entry.id !== deleteId));
      setToast({ message: 'Notification deleted successfully', type: 'success' });
    } catch (error: unknown) {
      setToast({ message: friendlyError(error, 'Failed to delete notification.'), type: 'error' });
    } finally {
      setDeleteId(null);
    }
  };

  const handleSend = async (): Promise<void> => {
    if (!sendTarget) return;
    setSending(true);
    try {
      const response = await fetchApi<NotificationSendResponse>(`/notifications/${sendTarget.id}/send`, { method: 'POST' });
      setNotifications((prev) =>
        prev.map((entry) => (entry.id === response.notification.id ? response.notification : entry))
      );
      const display = describeSendOutcome(response.result);
      setToast({ message: display.message, type: display.type });
    } catch (error: unknown) {
      setToast({ message: friendlyError(error, 'Failed to send notification.'), type: 'error' });
      const status = (error as { status?: unknown }).status;
      // 409 (already sent) / 404 (gone): refresh so the row reflects reality.
      if (status === 409 || status === 404) reload();
    } finally {
      setSending(false);
      setSendTarget(null);
    }
  };

  const filtered = notifications.filter((entry) => entry.title.toLowerCase().includes(search.toLowerCase()));

  const itemsPerPage = 10;
  const totalPages = Math.ceil(filtered.length / itemsPerPage);
  const currentPageSafe = Math.min(currentPage, Math.max(totalPages, 1));
  const paginatedData = filtered.slice((currentPageSafe - 1) * itemsPerPage, currentPageSafe * itemsPerPage);

  return (
    <div className={styles.container}>
      {toast && <Toast message={toast.message} type={toast.type} onClose={() => setToast(null)} />}

      <ConfirmModal
        isOpen={!!deleteId}
        title="Delete Notification"
        message="Are you sure you want to delete this notification/campaign? This action cannot be undone."
        onConfirm={handleDelete}
        onCancel={() => setDeleteId(null)}
        variant="danger"
        confirmText="Delete"
      />

      <ConfirmModal
        isOpen={!!sendTarget}
        title="Send Notification"
        message={
          sendTarget
            ? `â€œ${sendTarget.title}â€ will be delivered to the â€œ${segmentLabel(options, sendTarget.segmentId ?? sendTarget.targetSegment)}â€ segment on every registered active device. Continue?`
            : ''
        }
        onConfirm={handleSend}
        onCancel={() => setSendTarget(null)}
        variant="primary"
        confirmText="Send"
      />

      <div className={styles.header}>
        <div>
          <h1 className={styles.title}>Notifications & Campaigns</h1>
          <p className={styles.subtitle}>Manage push notification campaigns for Super Bae app users.</p>
        </div>
        <PermissionGate permission="NOTIFICATION_MANAGE">
          <Link href="/admin/notifications/create" className={styles.createBtn}>
            <Plus size={16} /> Create Notification
          </Link>
        </PermissionGate>
      </div>

      <div className={styles.toolbar}>
        <div className={styles.searchWrapper}>
          <Search size={16} className={styles.searchIcon} />
          <input
            type="text"
            placeholder="Search campaigns..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className={styles.searchInput}
          />
        </div>
        {optionsLoading && <span className={extra.hint}>Loading notification optionsâ€¦</span>}
      </div>

      {optionsError && !options && (
        <div className={extra.warningRow}>
          <span className={extra.warningText}>
            {optionsError} Segment labels may display as raw values until options load.
          </span>
          <button type="button" className={extra.retryBtn} onClick={retryOptions}>
            Retry
          </button>
        </div>
      )}

      <div className={styles.tableContainer}>
        {loading ? (
          <AdminTableSkeleton />
        ) : loadError ? (
          <div className={extra.errorBox}>
            <span className={extra.errorText}>{loadError}</span>
            <button type="button" className={extra.retryBtn} onClick={reload}>
              Retry
            </button>
          </div>
        ) : paginatedData.length === 0 ? (
          <NoData title="No notifications" description="No notifications or campaigns are available to display." />
        ) : (
          <>
            <table className={styles.table}>
              <thead>
                <tr>
                  <th>Title</th>
                  <th>Channel</th>
                  <th>Target Segment</th>
                  <th>Schedule</th>
                  <th>Status</th>
                  <th className={styles.actionsHeader}>Actions</th>
                </tr>
              </thead>
              <tbody>
                {paginatedData.map((notification) => (
                  <tr key={notification.id}>
                    <td>{notification.title}</td>
                    <td>{channelLabel(notification.channel ?? notification.type)}</td>
                    <td>{segmentLabel(options, notification.segmentId ?? notification.targetSegment)}</td>
                    <td>{formatSchedule(notification)}</td>
                    <td>
                      <span
                        className={`${styles.statusBadge} ${
                          notification.status === 'scheduled'
                            ? extra.scheduledBadge
                            : styles[statusBadgeModifier(notification.status)]
                        }`}
                      >
                        {notification.status}
                      </span>
                      {deliveryLine(notification) && (
                        <div
                          className={`${extra.delivery} ${notification.status === 'failed' ? extra.deliveryFailed : ''}`}
                        >
                          {deliveryLine(notification)}
                        </div>
                      )}
                    </td>
                    <td className={styles.actionsCell}>
                      <div className={styles.actionButtons}>
                        <PermissionGate permission="NOTIFICATION_MANAGE">
                          <Link
                            href={`/admin/notifications/${notification.id}`}
                            className={styles.iconBtn}
                            title="Edit"
                          >
                            <Edit2 size={16} />
                          </Link>
                          {canSend(notification, options) && (
                            <button
                              type="button"
                              className={styles.iconBtn}
                              title="Send"
                              disabled={sending}
                              onClick={() => setSendTarget(notification)}
                            >
                              <Send size={16} />
                            </button>
                          )}
                          <button
                            type="button"
                            className={`${styles.iconBtn} ${styles.danger}`}
                            onClick={() => setDeleteId(notification.id)}
                            title="Delete"
                          >
                            <Trash2 size={16} />
                          </button>
                        </PermissionGate>
                      </div>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
            {totalPages > 1 && (
              <div className={styles.pagination}>
                <button
                  className={styles.paginationBtn}
                  onClick={() => setCurrentPage((p) => Math.max(1, p - 1))}
                  disabled={currentPageSafe === 1}
                >
                  â† Previous
                </button>
                <span className={styles.paginationText}>
                  Page {currentPageSafe} of {totalPages}
                </span>
                <button
                  className={styles.paginationBtn}
                  onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))}
                  disabled={currentPageSafe === totalPages}
                >
                  Next â†’
                </button>
              </div>
            )}
          </>
        )}
      </div>
    </div>
  );
}
