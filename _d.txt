'use client';

import { useState, useEffect } from 'react';
import { useRouter, useParams } from 'next/navigation';
import Link from 'next/link';
import { ArrowLeft, CheckCircle, Send } from 'lucide-react';
import { fetchApi } from '../../../../lib/api/api';
import { Loader } from '../../../../components/admin/Loader';
import { Toast } from '../../../../components/admin/Toast';
import { PermissionGate } from '../../../../components/admin/PermissionGate';
import { ConfirmModal } from '../../../../components/admin/ConfirmModal';
import {
  canSend,
  channelLabel,
  describeSendOutcome,
  friendlyError,
  isFinalStatus,
  segmentLabel,
  statusSelectValues,
  toDatetimeLocal,
  type NotificationRecord,
  type NotificationSendResponse,
} from '../../../../lib/notifications/helpers';
import { useNotificationOptions } from '../../../../lib/notifications/options';
import styles from '../../admins/create/page.module.css';
import extra from '../notifications.module.css';

type ToastState = { message: string; type: 'success' | 'error' | 'warning' | 'info' } | null;

export default function EditNotificationPage() {
  const router = useRouter();
  const params = useParams();
  const id = params.id as string;

  const [loading, setLoading] = useState(true);
  const [loadAttempt, setLoadAttempt] = useState(0);
  const [loadError, setLoadError] = useState<string | null>(null);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [toast, setToast] = useState<ToastState>(null);

  const [notification, setNotification] = useState<NotificationRecord | null>(null);
  const [title, setTitle] = useState('');
  const [body, setBody] = useState('');
  const [channel, setChannel] = useState('push');
  const [segmentId, setSegmentId] = useState('all_users');
  const [scheduleAt, setScheduleAt] = useState('');
  const [status, setStatus] = useState('draft');
  const [scheduleError, setScheduleError] = useState<string | null>(null);

  const [sendConfirm, setSendConfirm] = useState(false);
  const [sending, setSending] = useState(false);

  const { options, loading: optionsLoading, error: optionsError, retry: retryOptions } = useNotificationOptions();

  // Load + prefill from the canonical fields (legacy aliases as fallback).
  // `loading` starts true and every setState happens after the awaited fetch.
  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const data = await fetchApi<NotificationRecord>(`/notifications/${id}`);
        if (cancelled) return;
        setNotification(data);
        setLoadError(null);
        setTitle(data.title || '');
        setBody(data.body || '');
        setChannel(String(data.channel ?? data.type ?? 'push'));
        const rawSegment = data.segmentId ?? data.targetSegment ?? 'all_users';
        setSegmentId(typeof rawSegment === 'string' && rawSegment ? rawSegment : 'all_users');
        setScheduleAt(toDatetimeLocal(data.scheduleAt ?? data.scheduledFor));
        setStatus(data.status || 'draft');
      } catch (error: unknown) {
        if (!cancelled) setLoadError(friendlyError(error, 'Failed to load notification.'));
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [id, loadAttempt]);

  const retryLoad = (): void => {
    setLoading(true);
    setLoadError(null);
    setLoadAttempt((current) => current + 1);
  };

  const statusReadOnly = isFinalStatus(status);
  const statusValues = statusSelectValues(status, options);
  const channelSendable = options ? options.sendableChannels.includes(channel) : false;

  const channelValues = options?.channels.length ? options.channels : [channel];
  const segmentOptions = options?.segments.length ? options.segments : [{ value: segmentId, label: segmentId }];
  const visibleSegments = segmentOptions.some((segment) => segment.value === segmentId)
    ? segmentOptions
    : [...segmentOptions, { value: segmentId, label: segmentId }];

  const handleScheduleChange = (value: string): void => {
    setScheduleAt(value);
    setScheduleError(null);
    // Keep status and schedule in sync like the create form does, so the
    // API never receives "scheduled" without a real date.
    if (value && status === 'draft') setStatus('scheduled');
    else if (!value && status === 'scheduled') setStatus('draft');
  };

  const handleSubmit = async (e: React.FormEvent): Promise<void> => {
    e.preventDefault();
    setScheduleError(null);
    const trimmedTitle = title.trim();
    const trimmedBody = body.trim();
    if (!trimmedTitle || !trimmedBody) {
      setToast({ message: 'Title and message body are required.', type: 'error' });
      return;
    }
    let scheduleIso: string | null = null;
    if (scheduleAt) {
      const date = new Date(scheduleAt);
      if (Number.isNaN(date.getTime())) {
        setScheduleError('Enter a valid schedule time.');
        return;
      }
      // Past dates are recorded history for sent/failed records; only an
      // active schedule must sit in the future.
      if (status === 'scheduled' && date.getTime() <= Date.now()) {
        setScheduleError('Schedule time must be in the future.');
        return;
      }
      scheduleIso = date.toISOString();
    }
    if (status === 'scheduled' && !scheduleIso) {
      setScheduleError('A scheduled notification needs a future date/time.');
      return;
    }
    setIsSubmitting(true);
    try {
      const updated = await fetchApi<NotificationRecord>(`/notifications/${id}`, {
        method: 'PATCH',
        body: JSON.stringify({
          title: trimmedTitle,
          body: trimmedBody,
          channel,
          segmentId,
          scheduleAt: scheduleIso,
          status,
        }),
      });
      setNotification(updated);
      setToast({ message: 'Notification updated successfully', type: 'success' });
      setTimeout(() => router.push('/admin/notifications'), 1500);
    } catch (error: unknown) {
      setToast({ message: friendlyError(error, 'Failed to update notification.'), type: 'error' });
      setIsSubmitting(false);
    }
  };

  const handleSend = async (): Promise<void> => {
    setSending(true);
    try {
      const response = await fetchApi<NotificationSendResponse>(`/notifications/${id}/send`, { method: 'POST' });
      setNotification(response.notification);
      setStatus(response.notification.status || status);
      setToast({ ...describeSendOutcome(response.result) });
    } catch (error: unknown) {
      setToast({ message: friendlyError(error, 'Failed to send notification.'), type: 'error' });
      const statusCode = (error as { status?: unknown }).status;
      // 409 already-sent / 404 missing: silently refresh so page state
      // matches the backend.
      if (statusCode === 409 || statusCode === 404) setLoadAttempt((current) => current + 1);
    } finally {
      setSending(false);
      setSendConfirm(false);
    }
  };

  if (loading) {
    return (
      <div className={styles.container}>
        <div style={{ display: 'flex', justifyContent: 'center', padding: 48 }}>
          <Loader />
        </div>
      </div>
    );
  }

  if (loadError || !notification) {
    return (
      <div className={styles.container}>
        <div className={styles.header}>
          <div style={{ display: 'flex', alignItems: 'center', gap: '16px' }}>
            <Link href="/admin/notifications" className={styles.backBtn}>
              <ArrowLeft size={20} />
            </Link>
            <div>
              <h1 className={styles.title}>Edit Notification</h1>
            </div>
          </div>
        </div>
        <div className={styles.content}>
          <div className={extra.errorBox}>
            <span className={extra.errorText}>{loadError ?? 'Notification not found.'}</span>
            <div style={{ display: 'flex', gap: 8, marginTop: 8 }}>
              <button type="button" className={extra.retryBtn} onClick={retryLoad}>
                Retry
              </button>
              <Link href="/admin/notifications" className={extra.retryBtn}>
                Back to list
              </Link>
            </div>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className={styles.container}>
      {toast && <Toast message={toast.message} type={toast.type} onClose={() => setToast(null)} />}

      <ConfirmModal
        isOpen={sendConfirm}
        title="Send Notification"
        message={`â€œ${title}â€ will be delivered to the â€œ${segmentLabel(options, segmentId)}â€ segment on every registered active device. Continue?`}
        onConfirm={handleSend}
        onCancel={() => setSendConfirm(false)}
        variant="primary"
        confirmText="Send"
      />

      <div className={styles.header}>
        <div style={{ display: 'flex', alignItems: 'center', gap: '16px' }}>
          <Link href="/admin/notifications" className={styles.backBtn}>
            <ArrowLeft size={20} />
          </Link>
          <div>
            <h1 className={styles.title}>Edit Notification</h1>
            <p className={styles.subtitle}>Update notification details, schedule, and delivery.</p>
          </div>
        </div>
      </div>

      <div className={styles.content}>
        <PermissionGate
          permission="NOTIFICATION_MANAGE"
          fallback={
            <div className={styles.errorAlert}>
              You do not have permission to manage notifications. Ask an administrator to grant the NOTIFICATION_MANAGE
              permission to your role.
              <div style={{ marginTop: 12 }}>
                <Link href="/admin/notifications" className={styles.cancelBtn}>
                  Back to Notifications
                </Link>
              </div>
            </div>
          }
        >
          <form onSubmit={handleSubmit} className={styles.form}>
            {optionsError && (
              <div className={extra.errorBox}>
                <span className={extra.errorText}>{optionsError}</span>
                <button type="button" className={extra.retryBtn} onClick={retryOptions}>
                  Retry
                </button>
              </div>
            )}

            <div className={styles.formSection}>
              <h2 className={styles.sectionTitle}>Notification Details</h2>
              <div className={styles.grid}>
                <div className={styles.field}>
                  <label className={styles.label}>Title</label>
                  <input
                    type="text"
                    value={title}
                    onChange={(e) => setTitle(e.target.value)}
                    className={styles.input}
                    required
                  />
                </div>

                <div className={styles.field}>
                  <label className={styles.label}>Channel</label>
                  <select
                    value={channel}
                    onChange={(e) => setChannel(e.target.value)}
                    className={styles.input}
                    disabled={optionsLoading}
                  >
                    {channelValues.map((value) => (
                      <option key={value} value={value}>
                        {channelLabel(value)}
                        {options && !options.sendableChannels.includes(value)
                          ? ' (record only â€” sending not implemented)'
                          : ''}
                      </option>
                    ))}
                  </select>
                  {optionsLoading && <span className={extra.hint}>Loading channelsâ€¦</span>}
                  {options && !channelSendable && (
                    <span className={`${extra.hint} ${extra.hintWarn}`}>
                      {channelLabel(channel)} delivery is not implemented yet â€” Send stays unavailable for this record.
                    </span>
                  )}
                </div>

                <div className={styles.field} style={{ gridColumn: '1 / -1' }}>
                  <label className={styles.label}>Message Body</label>
                  <textarea
                    value={body}
                    onChange={(e) => setBody(e.target.value)}
                    className={styles.input}
                    rows={4}
                    required
                  />
                </div>
              </div>
            </div>

            <div className={styles.formSection}>
              <h2 className={styles.sectionTitle}>Targeting & Scheduling</h2>
              <div className={styles.grid}>
                <div className={styles.field}>
                  <label className={styles.label}>Target Segment</label>
                  <select
                    value={segmentId}
                    onChange={(e) => setSegmentId(e.target.value)}
                    className={styles.input}
                    disabled={optionsLoading}
                  >
                    {visibleSegments.map((segment) => (
                      <option key={segment.value} value={segment.value}>
                        {segment.label}
                      </option>
                    ))}
                  </select>
                  {optionsLoading && <span className={extra.hint}>Loading segmentsâ€¦</span>}
                </div>

                <div className={styles.field}>
                  <label className={styles.label}>Schedule For</label>
                  <input
                    type="datetime-local"
                    value={scheduleAt}
                    onChange={(e) => handleScheduleChange(e.target.value)}
                    className={scheduleError ? `${styles.input} ${styles.inputError}` : styles.input}
                  />
                  {scheduleError ? (
                    <span className={styles.errorText}>{scheduleError}</span>
                  ) : (
                    <span className={extra.hint}>
                      A future time sets the status to Scheduled; the backend dispatcher delivers it at that time.
                    </span>
                  )}
                </div>

                <div className={styles.field}>
                  <label className={styles.label}>Status</label>
                  {statusReadOnly ? (
                    <div className={extra.deliveryRow}>
                      <span className={extra.finalBadge}>{status}</span>
                      <span className={extra.hint}>Sent is final â€” the delivery result below is recorded history.</span>
                    </div>
                  ) : (
                    <>
                      <select
                        value={status}
                        onChange={(e) => {
                          setStatus(e.target.value);
                          setScheduleError(null);
                        }}
                        className={styles.input}
                        disabled={optionsLoading}
                      >
                        {statusValues.map((value) => (
                          <option key={value} value={value}>
                            {value.charAt(0).toUpperCase() + value.slice(1)}
                          </option>
                        ))}
                      </select>
                      {optionsLoading && <span className={extra.hint}>Loading status optionsâ€¦</span>}
                      {!optionsLoading && !options && (
                        <span className={extra.hint}>Only the current status is available until options load.</span>
                      )}
                    </>
                  )}
                </div>
              </div>
            </div>

            <div className={styles.formSection}>
              <h2 className={styles.sectionTitle}>Delivery</h2>
              <div className={extra.deliveryRow}>
                {canSend(notification, options) ? (
                  <PermissionGate permission="NOTIFICATION_MANAGE">
                    <button
                      type="button"
                      className={styles.submitBtn}
                      onClick={() => setSendConfirm(true)}
                      disabled={sending || optionsLoading}
                    >
                      {sending ? (
                        <Loader />
                      ) : (
                        <>
                          <Send size={16} style={{ marginRight: '8px' }} /> Send now
                        </>
                      )}
                    </button>
                  </PermissionGate>
                ) : (
                  <span
                    className={`${extra.hint} ${statusReadOnly || channelSendable || !options ? '' : extra.hintWarn}`}
                  >
                    {statusReadOnly
                      ? 'This notification has already been sent â€” no further sends are possible.'
                      : options
                        ? `${channelLabel(channel)} delivery is not implemented yet.`
                        : 'Send becomes available once notification options have loaded.'}
                  </span>
                )}
                {status === 'failed' && options && channelSendable && (
                  <span className={extra.hint}>Previous delivery failed â€” use Send to retry.</span>
                )}
              </div>

              {notification.sendResult && (
                <div className={extra.resultPanel}>
                  <div className={extra.resultGrid}>
                    <div>
                      <div className={extra.resultLabel}>Outcome</div>
                      <div className={extra.resultValue}>{notification.sendResult.outcome ?? 'â€”'}</div>
                    </div>
                    <div>
                      <div className={extra.resultLabel}>Targeted</div>
                      <div className={extra.resultValue}>{notification.sendResult.targeted ?? 0}</div>
                    </div>
                    <div>
                      <div className={extra.resultLabel}>Delivered</div>
                      <div className={extra.resultValue}>{notification.sendResult.successful ?? 0}</div>
                    </div>
                    <div>
                      <div className={extra.resultLabel}>Failed</div>
                      <div className={extra.resultValue}>{notification.sendResult.failed ?? 0}</div>
                    </div>
                    <div>
                      <div className={extra.resultLabel}>Sent at</div>
                      <div className={extra.resultValue}>
                        {notification.sentAt ? new Date(notification.sentAt).toLocaleString() : 'â€”'}
                      </div>
                    </div>
                  </div>
                  {notification.sendResult.failures && notification.sendResult.failures.length > 0 && (
                    <ul className={extra.failureList}>
                      {notification.sendResult.failures.map((failure, index) => (
                        <li key={`${failure.tokenMask ?? 'device'}-${index}`} className={extra.failureItem}>
                          {failure.code ?? 'error'}: {failure.message ?? 'Delivery failed'} â€” device{' '}
                          {failure.tokenMask ?? 'unknown'}
                        </li>
                      ))}
                    </ul>
                  )}
                </div>
              )}
            </div>

            <div className={styles.formActions}>
              <Link href="/admin/notifications" className={styles.cancelBtn}>
                Cancel
              </Link>
              <button type="submit" className={styles.submitBtn} disabled={isSubmitting}>
                {isSubmitting ? (
                  <Loader />
                ) : (
                  <>
                    <CheckCircle size={16} style={{ marginRight: '8px' }} /> Save Changes
                  </>
                )}
              </button>
            </div>
          </form>
        </PermissionGate>
      </div>
    </div>
  );
}
